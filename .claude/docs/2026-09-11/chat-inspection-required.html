같이 정한 것이 코드에 어떻게 들어갔나

인수인계에서 남긴 약속은 하나였다. "명시적 이미지 조회 요청에서 재조회를 실제로 보장할 최소 방법을 찾고, 프롬프트만 늘려 해결을 주장하지 않는다."

그림 조회 요청에 도구 없이 답이 오면 한 번 재확인하고, 그래도 없으면 조회하지 않았다고 밝힌다

같다

약속: Artist에서 특정 모습의 그림을 직접 조회해 달라고 하면, 화면 요약에 그림이 없다고 적혀 있어도 저장소를 실제로 조회한 뒤 답한다. 끝내 조회하지 않았으면 조회하지 않았다고 밝힌다.

텍스트만 온 응답에서, 조회 요청인데 inspect_project 결과가 없으면 재확인 메시지를 붙여 한 번 더 요청한다. 두 번째에도 없으면 안내를 덧붙여 반환한다.

export function requestsImageInspection(stage: string, message: string): boolean {
  if (stage !== 'artist') return false
  return message.split(/[.。!?\n]/).some(sentence =>
    !/(하지\s*마|말고|말아|필요\s*없|안\s*봐도|don't|do not|no need)/i.test(sentence) &&
    !/(생성|만들|그려|generate|create|draw)/i.test(sentence) &&
    /(이미지|그림|시트|사진|image|picture|sheet|photo)/i.test(sentence) &&
    /(조회|확인|판독|비교|보고|봐|보여|살펴|\b(look|inspect|check|view|compare|see)\b)/i.test(sentence))
}
